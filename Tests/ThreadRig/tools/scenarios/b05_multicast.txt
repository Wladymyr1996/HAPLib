# B5 multicast: every other board receives it exactly once, the sender never.
# Repeated right after C rebooted, to cover a node that has only just attached
# (still a REED when it joins).
# Setup: wait for attachment, then remember every board's id and address.
until @A !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab A_ID @A /HAPINFO id=(\w+)/
grab A_ADDR @A /addr=(\S+)/
grab A_HEAP0 @A /heap=(\d+)/
until @B !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab B_ID @B /HAPINFO id=(\w+)/
grab B_ADDR @B /addr=(\S+)/
grab B_HEAP0 @B /heap=(\d+)/
until @C !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab C_ID @C /HAPINFO id=(\w+)/
grab C_ADDR @C /addr=(\S+)/
grab C_HEAP0 @C /heap=(\d+)/

mark m1
@A !hap mcast m
expect @A /HAPTX ok=1/ since=m1
count @B /HAPRX from=${A_ADDR} len=1 seq=- fill=- data=m/ n=1 timeout=5 since=m1
count @C /HAPRX from=${A_ADDR} len=1 seq=- fill=- data=m/ n=1 timeout=5 since=m1
absent @A /HAPRX/ window=1 since=m1

# C reboots; send the moment it reports attached
@C reset
expect @C /HAPBOOT id=${C_ID}/ timeout=30
expect @C /HAPSTATE attached=1/ timeout=40
mark m2
@A !hap mcast m2
expect @A /HAPTX ok=1/ since=m2
count @C /HAPRX from=${A_ADDR} len=2 seq=- fill=- data=m2/ n=1 timeout=5 since=m2
count @B /HAPRX from=${A_ADDR} len=2 seq=- fill=- data=m2/ n=1 timeout=5 since=m2
absent @A /HAPRX/ window=1 since=m2
