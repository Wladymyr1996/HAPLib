# B7 reboot identity: after `ot reset` B keeps its id and its ML-EID, and A to B
# works with the same address.
# Setup: wait for attachment, then remember every board's id and address.
until @A !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab A_ID @A /HAPINFO id=(\w+)/
grab A_ADDR @A /addr=(\S+)/
grab A_HEAP0 @A /heap=(\d+)/
until @B !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab B_ID @B /HAPINFO id=(\w+)/
grab B_ADDR @B /addr=(\S+)/
grab B_HEAP0 @B /heap=(\d+)/
until @C !hap info => /HAPINFO id=\w+ addr=\S+ attached=1/ timeout=30
grab C_ID @C /HAPINFO id=(\w+)/
grab C_ADDR @C /addr=(\S+)/
grab C_HEAP0 @C /heap=(\d+)/

@B reset
expect @B /HAPBOOT id=${B_ID}/ timeout=30
expect @B /HAPSTATE attached=1/ timeout=40
# the console is still printing its banner right after boot: let it finish, and
# poll, because the node first attaches as a leader of its own partition and
# merges back a little later
sleep 3
until @B !hap info => /HAPINFO id=${B_ID} addr=${B_ADDR} attached=1/ timeout=30

mark m1
@A !hap send ${B_ADDR} again
expect @A /HAPTX ok=1/ since=m1
count @B /HAPRX from=${A_ADDR} len=5 seq=- fill=- data=again/ n=1 timeout=5 since=m1
